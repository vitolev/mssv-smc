\begin{tikzpicture}[
    particle/.style={circle, draw, minimum size=1.15cm, inner sep=0pt},
    alive/.style={particle, fill=gray!20},
    dead/.style={particle, dashed, draw=gray, text=gray},
    onpath/.style={particle, fill=blue!18, draw=blue!60!black, line width=1.1pt},
    edge/.style={-{Stealth[scale=1.3]}, gray!70!black},
    pathedge/.style={-{Stealth[scale=1.3]}, blue!60!black, line width=1.3pt},
]
    % Particles: "alive" particles have descendants at the final time,
    % "dead" particles do not, "onpath" marks the highlighted trajectory.
    \foreach \t/\j/\s in {
        0/1/dead, 0/2/alive, 0/3/dead, 0/4/onpath, 0/5/alive,
        1/1/dead, 1/2/alive, 1/3/dead, 1/4/onpath, 1/5/alive,
        2/1/alive, 2/2/dead, 2/3/dead, 2/4/onpath, 2/5/alive,
        3/1/alive, 3/2/alive, 3/3/onpath, 3/4/alive, 3/5/alive} {
        \node[\s] (X\t\j) at (4*\t,-1.5*\j) {$X_{\t}^{(\j)}$};
    }

    % Edges: particle i at time t is propagated from its ancestor A_{t-1}^{(i)}.
    \foreach \t/\i/\a/\s in {
        1/1/1/edge, 1/2/2/edge, 1/3/3/edge, 1/4/4/pathedge, 1/5/5/edge,
        2/1/2/edge, 2/2/2/edge, 2/3/2/edge, 2/4/4/pathedge, 2/5/5/edge,
        3/1/1/edge, 3/2/1/edge, 3/3/4/pathedge, 3/4/4/edge, 3/5/5/edge} {
        \pgfmathtruncatemacro{\p}{\t-1}
        \draw[\s] (X\p\a.east) -- (X\t\i.west);
    }

    % Time labels.
    \foreach \t in {0,1,2,3} {
        \node[above, yshift=0.35cm] at (X\t1.north) {$t=\t$};
    }

    % Ancestor indices and offspring counts of each resampling step.
    \foreach \t/\anc/\off in {
        0/{1,2,3,4,5}/{1,1,1,1,1},
        1/{2,2,2,4,5}/{0,3,0,1,1},
        2/{1,1,4,4,5}/{2,0,0,2,1}} {
        \node[below, align=center, font=\small] at (4*\t+2,-8.3)
            {$A_{\t}^{(1:5)} = (\anc)$\\[2pt] $N_{\t}^{(1:5)} = (\off)$};
    }
\end{tikzpicture}
